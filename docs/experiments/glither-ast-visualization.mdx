import { useState } from 'react';

{/* AstNode: collapsible tree node with aligned connector lines */}
export function AstNode({ node, depth = 0 }) {
  const [collapsed, setCollapsed] = useState(false);
  const hasChildren = node.children?.length > 0;
  const toggleWidth = 14;

  const colors = {
    root:      '#f3c46c',
    keyword:   '#7ab8ff',
    structure: '#8ec07c',
    value:     '#b8bb26',
    action:    '#f2a77a',
    predicate: '#d3869b',
    meta:      '#9f9f9f',
  };
  const color = colors[node.role] || colors.structure;

  return (
    <div style={{ paddingLeft: depth > 0 ? 20 : 0 }}>
      <div
        onClick={hasChildren ? () => setCollapsed(c => !c) : undefined}
        style={{
          display: 'flex', alignItems: 'center', gap: 6,
          padding: '2px 0', cursor: hasChildren ? 'pointer' : 'default',
          userSelect: 'none',
        }}
      >
        <span style={{
          width: toggleWidth, display: 'inline-block', flexShrink: 0,
          color: '#737373', fontSize: 10, textAlign: 'center',
        }}>
          {hasChildren ? (collapsed ? '▶' : '▼') : ''}
        </span>
        <span style={{
          display: 'inline-block',
          background: color + '20',
          color: color,
          fontSize: 11,
          fontWeight: 600,
          fontFamily: 'monospace',
          padding: '0 6px',
          borderRadius: 3,
          border: '1px solid ' + color + '40',
          letterSpacing: '0.03em',
          textTransform: 'uppercase',
        }}>
          {node.type}
        </span>
        {node.value && (
          <span style={{
            color: '#d8c29d',
            fontSize: 12,
            fontFamily: 'monospace',
            marginLeft: 4,
          }}>
            {node.value}
          </span>
        )}
        {node.meta && (
          <span style={{
            color: '#737373',
            fontSize: 10,
            fontFamily: 'monospace',
            marginLeft: 6,
            fontStyle: 'italic',
          }}>
            {node.meta}
          </span>
        )}
      </div>
      {hasChildren && !collapsed && (
        <div style={{ borderLeft: '1px solid #2a2a2a', marginLeft: toggleWidth / 2 }}>
          {node.children.map((child, i) => (
            <AstNode key={i} node={child} depth={depth + 1} />
          ))}
        </div>
      )}
    </div>
  );
}

          flex: 1,
          background: '#0a0908',

  );
}

{/* AstLegend: color key */}
export function AstLegend() {
  const items = [
    { role: 'root',      label: 'root / module' },
    { role: 'keyword',   label: 'keyword / pragma' },
    { role: 'structure', label: 'structure / rule' },
    { role: 'value',     label: 'value / ident' },
    { role: 'action',    label: 'action / disposition' },
    { role: 'predicate', label: 'predicate / guard' },
    { role: 'meta',      label: 'metadata / comment' },
  ];
  return (
    <div style={{
      display: 'flex', flexWrap: 'wrap', gap: 8,
      padding: '10px 14px',
      border: '1px solid #2a2a2a',
      borderRadius: 6,
      background: '#121212',
      marginBottom: 24,
      fontFamily: 'monospace',
    }}>
      <span style={{ color: '#737373', fontSize: 10, textTransform: 'uppercase', letterSpacing: '0.06em' }}>AST node legend:</span>
      {items.map(i => (
        <span key={i.role} style={{
          fontSize: 10,
          display: 'inline-flex', alignItems: 'center', gap: 4,
        }}>
          <span style={{
            display: 'inline-block',
            width: 8, height: 8, borderRadius: 2,
            background: (() => {
              const colors = { root:'#f3c46c', keyword:'#7ab8ff', structure:'#8ec07c', value:'#b8bb26', action:'#f2a77a', predicate:'#d3869b', meta:'#9f9f9f' };
              return colors[i.role];
            })(),
          }} />
          <span style={{ color: '#d8c29d' }}>{i.label}</span>
        </span>
      ))}
    </div>
  );
}

## Mailguard rule (glither.mail dialect)

The canonical `mailguard.glith` ruleset — a flat mail-policy DSL that drops DMARC-reject mail and quarantines suspicious senders.

```glither
#pragma dialect glither.mail ; fold first-match

rule drop_dmarc_reject =
  msg | from.domain != dkim.domain
      | not sender.allowlisted
      | dmarc.policy == reject
  => into dropped (reason = dmarc_reject)

rule quarantine_spoof =
  msg | from.domain != dkim.domain
      | not sender.allowlisted
  => hold into quarantine
       on    admin.release   into delivered
       after 72h             into dropped (reason = quarantine_timeout)
```

## Glither rule (Roux core grammar)

A richer rule with a group definition, regex predicates, and a conditional arm. This exercises more of the Roux PEG grammar.

```glither
#pragma dialect glither.core

group high_risk = msg | dmarc.policy == reject
                     | spf.result == fail

rule route_high_risk =
  msg | from.domain ~ /(bank|pay|sso)\.(com|org|io)$/
      | not recipient.org == from.domain
  => raise clearance to restricted
     for recipient.org
  , route to admin_hold
  when dmarc.policy == reject
  after 24h into dropped (reason = unclaimed)
```

## Gleam — the auditable oracle

Gleam functions serve as the "auditable oracle" — a human-readable rendering of the compiled IR that is checked for conformance against the WASM output.

```gleam
pub type Clearance {
  Unrestricted
  Restricted
  Confidential
  Secret
  TopSecret
}

pub type Receipt {
  Receipt(outcome: Outcome, by: String, ts: Int)
}

pub fn dispose(seg: Segment, lig: Ligands) -> Receipt {
  case seg.scope, seg.detail, lig.clearance, lig.lineage.verified {
    "security/" <> _, Full, c, True if c >= Restricted ->
      seg
      |> wrap_to(SecurityCleared)
      |> seed
      |> receipt(Shadowed, by: "restricted_security_full")

    _, _, _, _ ->
      receipt(Denied, by: "default_deny")
  }
}
```

## How the AST maps to the compiler pipeline

The parse trees above correspond to the first stage of the Glither compiler pipeline:

<pre style={{
  background: '#0a0908',
  border: '1px solid #2a2a2a',
  borderRadius: 6,
  padding: 14,
  fontSize: 12,
  fontFamily: 'monospace',
  color: '#d8c29d',
  lineHeight: 1.7,
}}>
<span style={{color:'#7ab8ff', fontWeight:600}}>source</span> (.glith / .gleam)
  │
  ▼
<span style={{color:'#f3c46c', fontWeight:600}}>PEG parse</span> (roux.pest / gleam parser)
  │  ◀── <i style={{color:'#737373'}}>the AST trees above</i>
  ▼
<span style={{color:'#f3c46c', fontWeight:600}}>lower</span> → typed core IR (Roux IR nodes)
  │
  ▼
<span style={{color:'#f3c46c', fontWeight:600}}>infer</span> → phase-annotated IR
  │
  ▼
<span style={{color:'#f3c46c', fontWeight:600}}>codegen</span> → WASM component (via Rust) / Gleam package (oracle)
  │                 │
  ▼                 ▼
<span style={{color:'#8ec07c'}}>component.wasm</span>     <span style={{color:'#8ec07c'}}>target/glither-gen/*.gleam</span>
</pre>

The AST is produced by the **PEG parse** stage (roux.pest for Glither, or the Gleam parser for oracle code). The `lower` pass transforms the concrete parse tree into the typed core IR — discarding whitespace and comment nodes, resolving references, and annotating each node with WIT types.

See [glither WASM/WIT compiler spec](./glither-wasm-wit-compiler-spec.md) for the full compiler architecture, and [glither mailguard example](./glither-mailguard-example.md) for an end-to-end worked example through the pipeline.
